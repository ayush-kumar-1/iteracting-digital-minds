\newenvironment{quotebox}[1]
{%
  \def\quoteboxcitation{#1}%
  \par\smallskip
  \noindent
  \begin{tcolorbox}[
    enhanced,
    colback=QuoteBoxBg,
    colframe=DeckInk,
    boxrule=0.7pt,
    arc=10pt,
    width=\textwidth,
    left=12pt,
    right=12pt,
    top=10pt,
    bottom=10pt,
    boxsep=0pt,
    before skip=0pt,
    after skip=0pt
  ]
    \raggedright
    \small
    ``%
}
{%
    ''\par
    \vspace{0.6em}
    {\raggedleft\normalfont\small\quoteboxcitation\par}
  \end{tcolorbox}
  \par\smallskip
}
